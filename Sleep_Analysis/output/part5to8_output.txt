> check_nonlinear <- function(outcome, pred_ordered, 
+     data) {
+     d <- data
+     d$pred_n <- as.numeric(d[[pred_ordered]])
+     lin <- polr(reformulate("pred_n", response = outcome), data = d, 
+         Hess = TRUE)
+     quad <- polr(reformulate("poly(pred_n, 2)", response = outcome), 
+         data = d, Hess = TRUE)
+     cat(outcome, "~", pred_ordered, "\n")
+     cat("linear AIC:", AIC(lin), " | quadratic AIC:", AIC(quad), 
+         "\n\n")
+     list(lin = lin, quad = quad)
+ }
